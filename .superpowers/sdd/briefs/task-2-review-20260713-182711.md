import 'package:flutter/painting.dart';
import 'package:hanigold_admin/src/domain/users/model/list_transaction_info_item.model.dart';
import 'package:hanigold_admin/src/utils/data_table_text_widths.dart';
import 'package:hanigold_admin/src/utils/num_display.dart';
import 'package:persian_number_utility/persian_number_utility.dart';

enum ListUserInfoTxCol {
  row,
  name,
  cashBes,
  cashBed,
  goldBes,
  goldBed,
  coinBes,
  coinBed,
  currBes,
  currBed,
  totalBes,
  totalBed,
}

List<double> computeListUserInfoTransactionColumnWidths({
  required List<ListTransactionInfoItemModel> rows,
  required TextStyle headerStyle,
  required TextStyle bodyStyle,
  required TextStyle bodySmallStyle,
  double textScaleFactor = 1.0,
}) {
  final cellPad = kDataTableCellPad;
  final cellIconPad = kDataTableCellPad + kDataTableIconPad;

  double measure(String text, TextStyle style) =>
      measureTextWidth(text, style, textScaleFactor: textScaleFactor);

  void consider(List<double> maxes, int col, String text, TextStyle style) {
    final w = measure(text, style);
    if (w > maxes[col]) maxes[col] = w;
  }

  final maxes = List<double>.filled(ListUserInfoTxCol.values.length, 0);

  // Headers (match DataColumn labels; multi-part headers as single strings)
  consider(maxes, ListUserInfoTxCol.row.index, 'ردیف', headerStyle);
  consider(maxes, ListUserInfoTxCol.name.index, 'نام', headerStyle);
  consider(
    maxes,
    ListUserInfoTxCol.cashBes.index,
    'مانده ریالی(بستانکار)',
    headerStyle,
  );
  consider(
    maxes,
    ListUserInfoTxCol.cashBed.index,
    'مانده ریالی(بدهکار)',
    headerStyle,
  );
  consider(
    maxes,
    ListUserInfoTxCol.goldBes.index,
    'مانده طلا(بستانکار)',
    headerStyle,
  );
  consider(
    maxes,
    ListUserInfoTxCol.goldBed.index,
    'مانده طلا(بدهکار)',
    headerStyle,
  );
  consider(
    maxes,
    ListUserInfoTxCol.coinBes.index,
    'مانده سکه(بستانکار)',
    headerStyle,
  );
  consider(
    maxes,
    ListUserInfoTxCol.coinBed.index,
    'مانده سکه(بدهکار)',
    headerStyle,
  );
  consider(
    maxes,
    ListUserInfoTxCol.currBes.index,
    'مانده ارز(بستانکار)',
    headerStyle,
  );
  consider(
    maxes,
    ListUserInfoTxCol.currBed.index,
    'مانده ارز(بدهکار)',
    headerStyle,
  );
  consider(maxes, ListUserInfoTxCol.totalBes.index, 'تراز کل بس', headerStyle);
  consider(maxes, ListUserInfoTxCol.totalBed.index, 'تراز کل بد', headerStyle);

  for (final row in rows) {
    // row
    consider(
      maxes,
      ListUserInfoTxCol.row.index,
      '${row.rowNum ?? ''}',
      bodyStyle,
    );

    // name
    consider(
      maxes,
      ListUserInfoTxCol.name.index,
      row.accountName ?? '',
      bodyStyle,
    );

    // cashBes
    final cashBes = row.cashBalanceBes ?? 0;
    if (cashBes > 0) {
      consider(
        maxes,
        ListUserInfoTxCol.cashBes.index,
        'وجه نقد ${cashBes.toStringAsFixed(0).seRagham()} ریال',
        bodySmallStyle,
      );
    }
    for (final b in row.balances ?? const []) {
      if (b.unitName == 'ریال' && (b.balance ?? 0) > 0) {
        consider(
          maxes,
          ListUserInfoTxCol.cashBes.index,
          '${b.itemName ?? ''} ${b.balance!.toStringAsFixed(0).seRagham()} ریال',
          bodySmallStyle,
        );
      }
    }

    // cashBed
    final cashBed = row.cashBalanceBed ?? 0;
    if (cashBed < 0) {
      consider(
        maxes,
        ListUserInfoTxCol.cashBed.index,
        'وجه نقد -${cashBed.abs().toStringAsFixed(0).seRagham()} ریال',
        bodySmallStyle,
      );
    }
    for (final b in row.balances ?? const []) {
      if (b.unitName == 'ریال' && (b.balance ?? 0) < 0) {
        consider(
          maxes,
          ListUserInfoTxCol.cashBed.index,
          '${b.itemName ?? ''} -${b.balance!.abs().toStringAsFixed(0).seRagham()} ریال',
          bodySmallStyle,
        );
      }
    }

    // goldBes
    final goldBes = row.goldBalanceBes ?? 0;
    if (goldBes > 0) {
      consider(
        maxes,
        ListUserInfoTxCol.goldBes.index,
        'آبشده ${goldBes.toStringAsFixed(3)} گرم',
        bodySmallStyle,
      );
    }
    for (final b in row.balances ?? const []) {
      if (b.unitName == 'گرم' && (b.balance ?? 0) > 0) {
        consider(
          maxes,
          ListUserInfoTxCol.goldBes.index,
          '${b.itemName ?? ''} ${b.balance} گرم',
          bodySmallStyle,
        );
      }
    }

    // goldBed
    final goldBed = row.goldBalanceBed ?? 0;
    if (goldBed < 0) {
      consider(
        maxes,
        ListUserInfoTxCol.goldBed.index,
        'آبشده -${goldBed.abs().toStringAsFixed(3)} گرم',
        bodySmallStyle,
      );
    }
    for (final b in row.balances ?? const []) {
      if (b.unitName == 'گرم' && (b.balance ?? 0) < 0) {
        consider(
          maxes,
          ListUserInfoTxCol.goldBed.index,
          '${b.itemName ?? ''} -${b.balance!.abs()} گرم',
          bodySmallStyle,
        );
      }
    }

    // coinBes
    if ((row.coinBalanceBes ?? 0) != 0) {
      consider(
        maxes,
        ListUserInfoTxCol.coinBes.index,
        'تمام سکه ${row.coinBalanceBes!.toDisplayString()} عدد',
        bodySmallStyle,
      );
      consider(
        maxes,
        ListUserInfoTxCol.coinBes.index,
        'نیم سکه ${row.halfCoinBalanceBes?.toDisplayString() ?? ''} عدد',
        bodySmallStyle,
      );
      consider(
        maxes,
        ListUserInfoTxCol.coinBes.index,
        'ربع سکه ${row.quarterCoinBalanceBes?.toDisplayString() ?? ''} عدد',
        bodySmallStyle,
      );
    }

    // coinBed
    if ((row.coinBalanceBed ?? 0) != 0) {
      consider(
        maxes,
        ListUserInfoTxCol.coinBed.index,
        'تمام سکه -${row.coinBalanceBed!.abs().toDisplayString()} عدد',
        bodySmallStyle,
      );
      consider(
        maxes,
        ListUserInfoTxCol.coinBed.index,
        'نیم سکه -${row.halfCoinBalanceBed?.abs().toDisplayString() ?? ''} عدد',
        bodySmallStyle,
      );
      consider(
        maxes,
        ListUserInfoTxCol.coinBed.index,
        'ربع سکه -${row.quarterCoinBalanceBed?.abs().toDisplayString() ?? ''} عدد',
        bodySmallStyle,
      );
    }

    // currBes / currBed (دلار lines)
    for (final b in row.balances ?? const []) {
      if (b.unitName != 'دلار' || b.balance == null) continue;
      if (b.balance! > 0) {
        consider(
          maxes,
          ListUserInfoTxCol.currBes.index,
          ' ${b.itemName} ${b.balance!.toDisplayString()} ${b.unitName} ',
          bodySmallStyle,
        );
      } else if (b.balance! < 0) {
        consider(
          maxes,
          ListUserInfoTxCol.currBed.index,
          ' ${b.itemName} ${b.balance!.toDisplayString()} ${b.unitName} ',
          bodySmallStyle,
        );
      }
    }

    // totalBes
    final currencyBes = row.currencyValueBes ?? 0;
    if (currencyBes > 0) {
      consider(
        maxes,
        ListUserInfoTxCol.totalBes.index,
        '${currencyBes.toStringAsFixed(0).seRagham()} ریال',
        bodyStyle,
      );
      final gv = row.goldValue ?? 0;
      final goldPart = gv < 0
          ? '-${gv.abs().toStringAsFixed(3)}'
          : ' ${gv.toStringAsFixed(3)} ';
      consider(
        maxes,
        ListUserInfoTxCol.totalBes.index,
        ' معادل آبشده : $goldPart گرم ',
        bodySmallStyle,
      );
    }

    // totalBed
    final currencyBed = row.currencyValueBed ?? 0;
    if (currencyBed < 0) {
      consider(
        maxes,
        ListUserInfoTxCol.totalBed.index,
        '-${currencyBed.abs().toStringAsFixed(0).seRagham()} ریال',
        bodyStyle,
      );
      final gv = row.goldValue ?? 0;
      final goldPart = gv < 0
          ? '-${gv.abs().toStringAsFixed(3)}'
          : ' ${gv.toStringAsFixed(3)} ';
      consider(
        maxes,
        ListUserInfoTxCol.totalBed.index,
        ' معادل آبشده : $goldPart گرم ',
        bodySmallStyle,
      );
    }
  }

  return [
    clampColumnWidth(
      maxes[ListUserInfoTxCol.row.index],
      minWidth: 40,
      maxWidth: 80,
      padding: cellPad,
    ),
    clampColumnWidth(
      maxes[ListUserInfoTxCol.name.index],
      minWidth: 72,
      maxWidth: 220,
      padding: cellPad,
    ),
    clampColumnWidth(
      maxes[ListUserInfoTxCol.cashBes.index],
      minWidth: 100,
      maxWidth: 280,
      padding: cellIconPad,
    ),
    clampColumnWidth(
      maxes[ListUserInfoTxCol.cashBed.index],
      minWidth: 100,
      maxWidth: 280,
      padding: cellIconPad,
    ),
    clampColumnWidth(
      maxes[ListUserInfoTxCol.goldBes.index],
      minWidth: 100,
      maxWidth: 260,
      padding: cellIconPad,
    ),
    clampColumnWidth(
      maxes[ListUserInfoTxCol.goldBed.index],
      minWidth: 100,
      maxWidth: 260,
      padding: cellIconPad,
    ),
    clampColumnWidth(
      maxes[ListUserInfoTxCol.coinBes.index],
      minWidth: 100,
      maxWidth: 240,
      padding: cellPad,
    ),
    clampColumnWidth(
      maxes[ListUserInfoTxCol.coinBed.index],

      maxes[ListUserInfoTxCol.currBes.index],

  ];
}

import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:hanigold_admin/src/domain/users/model/list_transaction_info_item.model.dart';
import 'package:hanigold_admin/src/domain/users/view/list_user_info_transaction_column_widths.dart';
import 'package:hanigold_admin/src/utils/data_table_text_widths.dart';

ListTransactionInfoItemModel _row({
  String name = 'A',
  int rowNum = 1,
  double? cashBes,
}) =>
    ListTransactionInfoItemModel(
      accountId: 1,
      accountName: name,
      currencyValueBes: 0,
      currencyValueBed: 0,
      goldBalanceBes: 0,
      goldBalanceBed: 0,
      coinBalanceBes: 0,
      coinBalanceBed: 0,
      halfCoinBalanceBes: 0,
      halfCoinBalanceBed: 0,
      quarterCoinBalanceBes: 0,
      quarterCoinBalanceBed: 0,
      cashBalanceBes: cashBes ?? 0,
      cashBalanceBed: 0,
      goldValue: 0,
      coinValue: 0,
      afterGoldBalance: 0,
      afterCashBalance: 0,
      rowNum: rowNum,
      balances: const [],
    );

void main() {
  TestWidgetsFlutterBinding.ensureInitialized();

  test('longer text measures wider than shorter text', () {
    const style = TextStyle(fontSize: 12, fontFamily: 'IranSansR');
    final shortW = measureTextWidth('نام', style);
    final longW = measureTextWidth('نام حساب بسیار طولانی برای تست', style);
    expect(longW, greaterThan(shortW));
  });

  test('clampColumnWidth respects min and max', () {
    expect(clampColumnWidth(10, minWidth: 40, maxWidth: 200), 40);
    expect(clampColumnWidth(500, minWidth: 40, maxWidth: 200), 200);
    expect(clampColumnWidth(80, minWidth: 40, maxWidth: 200, padding: 8), 88);
  });

  test('name column grows with longer accountName', () {
    const header = TextStyle(fontSize: 11);
    const body = TextStyle(fontSize: 11);
    const small = TextStyle(fontSize: 9);
    final short = computeListUserInfoTransactionColumnWidths(
      rows: [_row(name: 'علی')],
      headerStyle: header,
      bodyStyle: body,
      bodySmallStyle: small,
    );
    final long = computeListUserInfoTransactionColumnWidths(
      rows: [_row(name: 'شرکت بازرگانی طلای نمونه با نام خیلی بلند')],
      headerStyle: header,
      bodyStyle: body,
      bodySmallStyle: small,
    );
    expect(long[ListUserInfoTxCol.name.index],
        greaterThan(short[ListUserInfoTxCol.name.index]));
  });

  test('returns exactly 12 widths', () {
    final w = computeListUserInfoTransactionColumnWidths(
      rows: const [],
      headerStyle: const TextStyle(fontSize: 11),
      bodyStyle: const TextStyle(fontSize: 11),
      bodySmallStyle: const TextStyle(fontSize: 9),
    );
    expect(w.length, 12);
  });
}
