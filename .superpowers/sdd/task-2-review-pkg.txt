/// Whether a pagehide event should trigger tab-close logout.
///
/// Skips when not web or when the page enters the back-forward cache
/// ([persisted] true) — session is preserved for bfcache restore.
bool shouldLogoutOnPageHide({
  required bool isWeb,
  required bool persisted,
}) {
  if (!isWeb) return false;
  if (persisted) return false;
  return true;
}

/// Logout-equivalent teardown for tab close: disconnect socket, clear session.
Future<void> performTabCloseLogout() async {
  try {
    if (Get.isRegistered<SocketService>()) {
      await SocketService.to.disconnect();
    }
  } catch (e, s) {
    AppLogger.e('Socket disconnect during tab close logout', e, s);
  }

  try {
    await clearStoredSession();
    clearSessionControllers();
  } catch (e, s) {
    AppLogger.e('Session clear during tab close logout', e, s);
  }
}

bool _readPageHidePersisted(html.Event event) {
  try {
    return (event as dynamic).persisted == true;
  } catch (_) {
    return false;
  }
}

/// Registers a [pagehide] handler that logs out when the tab is closed or navigated away.
void registerWebTabCloseLogout() {
  try {
    html.window.onPageHide.listen((event) {
      try {
        final persisted = _readPageHidePersisted(event);
        if (!shouldLogoutOnPageHide(isWeb: true, persisted: persisted)) {
          return;
        }
        unawaited(performTabCloseLogout());
      } catch (e, s) {
        AppLogger.e('pagehide tab-close logout handler failed', e, s);
      }
    });
  } catch (e, s) {
    AppLogger.e('registerWebTabCloseLogout failed', e, s);
  }
}
